```python
cache_patterns = ["__pycache__", ".pytest_cache", ".mypy_cache", ".ruff_cache"]

for pattern in cache_patterns:
    for cache_dir in Path(".").rglob(pattern):
        if should_exclude(cache_dir) or not cache_dir.is_dir():
            continue
        try:
            size = sum(f.stat().st_size for f in cache_dir.rglob("*") if f.is_file())
            shutil.rmtree(cache_dir)
            logger.info(f"Deleted cache directory: {cache_dir} ({size / 1024:.2f} KB)")
            stats["directories_removed"] += 1
            stats["bytes_freed"] += size
        except Exception as e:
            logger.error(f"Failed to delete {cache_dir}: {e}")
            stats["errors"].append(str(e))
```

```python
        if should_exclude(cache_file) or not cache_file.is_file():
```
